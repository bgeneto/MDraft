% XeLaTeX preview while the exported PDF has it.

\usepackage[T1]{fontenc}
\usepackage{mathptmx}
\begin{document}
Times text, set in the virtual font ptmr8t.

{\fontfamily{bch}\selectfont Charter text, set in bchr8t.}

A formula: $e^{i\pi} + 1 = 0$.
\end{document}
